    <script>
        // إعداد أسئلة جدول الضرب في 5
        const questions = [
            { num: 1, text: "5 × 1", answer: 5, options: [5, 10, 15, 0] },
            { num: 2, text: "5 × 2", answer: 10, options: [10, 15, 20, 5] },
            { num: 3, text: "5 × 3", answer: 15, options: [12, 15, 20, 25] },
            { num: 4, text: "5 × 4", answer: 20, options: [15, 20, 25, 30] },
            { num: 5, text: "5 × 5", answer: 25, options: [20, 25, 30, 35] },
            { num: 6, text: "5 × 6", answer: 30, options: [25, 30, 35, 40] },
            { num: 7, text: "5 × 7", answer: 35, options: [30, 35, 40, 45] },
            { num: 8, text: "5 × 8", answer: 40, options: [35, 40, 45, 50] },
            { num: 9, text: "5 × 9", answer: 45, options: [40, 45, 50, 55] },
            { num: 10, text: "5 × 10", answer: 50, options: [45, 50, 55, 60] }
        ];

        // احداثيات الصناديق والجزيرة النهائية
        const islandPositions = [
            { top: 75, right: 10 },  // 1
            { top: 55, right: 25 },  // 2
            { top: 75, right: 42 },  // 3
            { top: 50, right: 58 },  // 4
            { top: 25, right: 45 },  // 5
            { top: 20, right: 25 },  // 6
            { top: 15, right: 65 },  // 7
            { top: 35, right: 80 },  // 8
            { top: 65, right: 82 },  // 9
            { top: 80, right: 65 },  // 10
            { top: 45, right: 10 }   // الكنز النهائي 🏴‍☠️👑
        ];

        let studentName = "";
        let currentStep = 0;
        let score = 0;

        // بدء اللعبة
        function startGame() {
            const nameInput = document.getElementById('player-name-input').value.trim();
            if (!nameInput) {
                alert('لطفاً ادخل اسمك للبُدء بالمغامرة!');
                return;
            }
            studentName = nameInput;
            document.getElementById('start-screen').classList.add('hidden');
            initIslandMap();
            moveShipToNode(0);
        }

        // رسم الجزيرة والتنقيل
        function initIslandMap() {
            const container = document.getElementById('island-nodes-container');
            container.innerHTML = '';

            // إنشاء النقاط والصناديق
            for (let i = 0; i < questions.length; i++) {
                const pos = islandPositions[i];
                const node = document.createElement('div');
                node.id = `node-${i}`;
                node.className = `absolute island-node cursor-pointer flex flex-col items-center justify-center p-2 rounded-2xl bg-amber-100 border-2 border-amber-400 shadow-md ${i > 0 ? 'opacity-50 pointer-events-none' : ''}`;
                node.style.top = `${pos.top}%`;
                node.style.right = `${pos.right}%`;
                node.onclick = () => openQuestion(i);

                node.innerHTML = `
                    <div class="text-2xl md:text-3xl">📦</div>
                    <span class="text-[10px] font-black bg-amber-800 text-white px-1.5 py-0.5 rounded-full -mt-1">${i + 1}</span>
                `;
                container.appendChild(node);
            }

            // إضافة كنز الجزيرة النهائي
            const finalPos = islandPositions[10];
            const treasureNode = document.createElement('div');
            treasureNode.id = `node-final`;
            treasureNode.className = `absolute island-node opacity-40 pointer-events-none flex flex-col items-center justify-center p-3 rounded-full bg-yellow-300 border-4 border-yellow-500 shadow-2xl treasure-glow`;
            treasureNode.style.top = `${finalPos.top}%`;
            treasureNode.style.right = `${finalPos.right}%`;
            treasureNode.innerHTML = `
                <div class="text-4xl md:text-5xl">👑💎</div>
                <span class="text-xs font-black bg-red-600 text-white px-2 py-0.5 rounded-full mt-1">الكنز</span>
            `;
            container.appendChild(treasureNode);
        }

        // تحريك السفينة إلى موقع معين
        function moveShipToNode(index) {
            const ship = document.getElementById('ship');
            const pos = islandPositions[index];
            ship.style.top = `${pos.top}%`;
            ship.style.right = `${pos.right}%`;
        }

        // فتح صندوق السؤال
        function openQuestion(index) {
            if (index !== currentStep) return;

            const q = questions[index];
            document.getElementById('question-text').innerText = `${q.text} = ؟`;
            
            const optionsContainer = document.getElementById('options-container');
            optionsContainer.innerHTML = '';
            
            // خلط الخيارات
            const shuffledOptions = [...q.options].sort(() => Math.random() - 0.5);

            shuffledOptions.forEach(opt => {
                const btn = document.createElement('button');
                btn.className = "bg-amber-400 hover:bg-amber-500 text-amber-950 font-black text-2xl py-3 rounded-xl border-b-4 border-amber-600 shadow-md active:translate-y-1 transition-all";
                btn.innerText = opt;
                btn.onclick = () => checkAnswer(opt, q.answer, index);
                optionsContainer.appendChild(btn);
            });

            document.getElementById('feedback').innerText = '';
            document.getElementById('question-modal').classList.remove('hidden');
        }

        // التحقق من الإجابة
        function checkAnswer(selected, correct, index) {
            const feedback = document.getElementById('feedback');
            
            if (selected === correct) {
                feedback.className = "mt-4 text-lg font-bold text-emerald-600 animate-bounce";
                feedback.innerText = "إجابة صحيحة! أحسنت يا بطل! 🌟";
                
                score += 10;
                document.getElementById('score').innerText = score;

                // تحديث حالة الصندوق الحالي
                const currentNode = document.getElementById(`node-${index}`);
                currentNode.innerHTML = `<div class="text-2xl md:text-3xl">✨🔓</div>`;
                currentNode.classList.add('bg-emerald-200', 'border-emerald-500');

                setTimeout(() => {
                    document.getElementById('question-modal').classList.add('hidden');
                    currentStep++;
                    document.getElementById('progress').innerText = `${currentStep} / 10`;

                    if (currentStep < questions.length) {
                        // تفعيل الصندوق التالي
                        const nextNode = document.getElementById(`node-${currentStep}`);
                        nextNode.classList.remove('opacity-50', 'pointer-events-none');
                        moveShipToNode(currentStep);
                    } else {
                        // الوصول للكنز النهائي!
                        moveShipToNode(10);
                        const finalNode = document.getElementById('node-final');
                        finalNode.classList.remove('opacity-40');
                        
                        setTimeout(showCertificate, 1500);
                    }
                }, 1000);

            } else {
                feedback.className = "mt-4 text-lg font-bold text-rose-600 animate-pulse";
                feedback.innerText = "حاول مرة أخرى يا بطل! 💡";
            }
        }

        // عرض شهادة التقدير وإطلاق الألعاب النارية
        function showCertificate() {
            document.getElementById('cert-student-name').innerText = studentName;
            document.getElementById('cert-score').innerText = `${score} / 100`;
            document.getElementById('certificate-modal').classList.remove('hidden');

            // إطلاق الاحتفال بالورود والألوان
            if (typeof confetti === 'function') {
                confetti({
                    particleCount: 120,
                    spread: 80,
                    origin: { y: 0.6 }
                });
            }
        }

        // إعادة اللعبة
        function restartGame() {
            currentStep = 0;
            score = 0;
            document.getElementById('score').innerText = '0';
            document.getElementById('progress').innerText = '0 / 10';
            document.getElementById('certificate-modal').classList.add('hidden');
            initIslandMap();
            moveShipToNode(0);
        }
    </script>
